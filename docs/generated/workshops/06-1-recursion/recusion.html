def visualize_tree(tree):
    def add_nodes_edges(tree, dot=None):
        # สร้าง Digraph object
        if dot is None:
            dot = Digraph()
            dot.node(name=str(tree), label=str(tree.val))

        # เพิ่ม nodes
        if tree.left:
            dot.node(name=str(tree.left) ,label=str(tree.left.val))
            dot.edge(str(tree), str(tree.left))
            dot = add_nodes_edges(tree.left, dot=dot)

        if tree.right:
            dot.node(name=str(tree.right) ,label=str(tree.right.val))
            dot.edge(str(tree), str(tree.right))
            dot = add_nodes_edges(tree.right, dot=dot)

        return dot

    # เพิ่ม nodes วนไปเรื่อยๆ และสร้าง list ของ edges
    dot = add_nodes_edges(tree)

    # Visualize the graph
    display(dot)

def fibonacci(value):

  ## เติมโค้ดด้านล่าง ##

  if value <= 1: return value,TreeNode(value)
  tree = TreeNode(value)
  left_value, left_t = fibonacci(value-1)
  right_value, right_t = fibonacci(value-2)
  sum_value = left_value + right_value
  tree.left = left_t
  tree.right = right_t

  ##############

  return sum_value,tree

# Edit Distance
ในส่วนนี้ เราต้องการที่จะสร้างฟังก์ชั่นที่ใช้เช็คความแตกต่างของประโยค 2 ประโยค --- เราต้องสร้าง **Minimum Edit Distance function** เพื่อคำนวณหาค่าความแตกต่างระหว่างข้อความ 2 อัน \\ ในส่วนของการทำงานของ Minimum Edit Distance นั้น, สมมุติว่าเรามีสตริง 2 สตริง \\ **m** คือ ความยาวของสตริงที่ 1 \\ **n** คือ ความยาวของสตริงที่ 2 เราจะสนใจทุกๆการเปรียบเทียบบนตัวอักษรตัวสุดท้ายของสตริงที่ 1 โดยเราจะคำนวณวนไปเรื่อยๆเพื่อ**หาค่าที่น้อยที่สุดของการเปรียบเทียบทั้ง 3 แบบ** \\ โดยการเปรียบเทียบทั้ง 3 แบบได้แก่ * **การเติม: เราจะให้ฟังก์ชั่นรับ m และ n-1** * **การลบ: เราจะให้ฟังก์ชั่นรับ m-1 และ n** * **การแทนที่: เราจะให้ฟังก์ชั่นรับ m-1 และ n-1** ในกรณี **สตริงเปล่าเทียบกับสตริงอื่นให้คืนความยาวของสตริงที่ไม่ใช่สตริงเปล่า** กลับมา **ย้ำาา: ทุกๆการเปรียบเทียบจะมีค่าการทำงานเท่ากับ 1 !!**

# TODO 2.1 สร้างฟังก์ชั่น recursive ของ Minimum Edit Distance
def edit_distance(real,fake,m,n):

  if m == 0:
    return n #เติมสิ่งที่ฟังก์ชั่นต้องคืนกลับมา เมื่อสตริงตัวที่ 1 ที่ให้เช็คเป็นสตริงเปล่า

  if n == 0:
    return m #เติมสิ่งที่ฟังก์ชั่นต้องคืนกลับมา เมื่อสตริงตัวที่ 2 ที่ให้เช็คเป็นสตริงเปล่า

  #เติมสิ่งที่ฟังก์ชั่นต้องคืนกลับมา ในกรณีที่ตัวสุดท้ายของสตริงทั้ง 2 อันเหมือนกันให้ข้ามไปคำนวณตัวถัดไป
  if real[m-1] == fake[n-1]:
    return edit_distance(real,fake,m-1,n-1)


  #เติมสิ่งที่ฟังก์ชั่นต้องคืนค่าให้สมบูรณ์ เพื่อที่ฟังก์ชั่นจะได้ทำงานถูกต้อง
  return min(edit_distance(real,fake,m,n-1),edit_distance(real,fake,m-1,n),edit_distance(real,fake,m-1,n-1)) + 1
